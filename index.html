using UnityEngine; using UnityEngine.EventSystems; using System.Collections; namespace ModernHorrorGame { // --- ИНТЕРФЕЙС ВЗАИМОДЕЙСТВИЯ --- public interface IInteractable { void Interact(); } // --- ОСНОВНОЙ КОНТРОЛЛЕР ИГРОКА И МЕНЕДЖЕР СЮЖЕТА --- [RequireComponent(typeof(CharacterController))] public class GameDirector : MonoBehaviour, IDragHandler, IPointerDownHandler, IPointerUpHandler, IInteractable { public static GameDirector Instance; [Header("Scene References")] public Transform cameraTransform; public Light roomLight; public AudioSource ambienceSource; public AudioSource sfxSource; public GameObject interactionButton; [Header("Room Objects")] public GameObject tvStaticScreen; public GameObject tvContentScreen; public Transform closetHidePoint; public GameObject fuseBoxObject; [Header("Audio Tracks")] public AudioClip tvStaticClip; public AudioClip scaryBuildupClip; public AudioClip switchClickClip; public AudioClip whisperClip; [Header("Movement & Look Settings")] public float moveSpeed = 3.5f; public float lookSensitivity = 1.5f; // Внутренние состояния private CharacterController controller; private Vector2 moveInput; private Vector2 lookInput; private float verticalRotation = 0f; private bool isInteracting = false; private GameObject currentInteractable; private bool isPowerOut = false; private bool isHiding = false; // Состояния для динамических объектов (выключатель, шкаф, счетчик) [HideInInspector] public enum ObjectType { LightSwitch, Closet, FuseBox, TV } public ObjectType objectType; // Используется, если этот же скрипт вешается на объекты комнаты private void Awake() { if (Instance == null) Instance = this; controller = GetComponent(); } private void Start() { // Если это главный менеджер сцены if (gameObject.CompareTag("Player") || controller != null) { InitializeGame(); StartCoroutine(MasterHorrorSequence()); } } private void Update() { // Управление работает только если игрок не прячется if (!isHiding && controller != null) { HandleMovement(); HandleLook(); CheckInteractables(); } } # ========================================== # 1. МОБИЛЬНОЕ УПРАВЛЕНИЕ И ВЗАИМОДЕЙСТВИЕ # ========================================== public void OnPointerDown(PointerEventData eventData) { } public void OnDrag(PointerEventData eventData) { // Левая половина экрана — джойстик движения, правая — обзор (свайп) if (eventData.position.x > Screen.width * 0.5f) { lookInput = eventData.delta * lookSensitivity * 0.05f; } else { moveInput = (eventData.position - eventData.pressPosition).normalized; moveInput = Vector2.ClampMagnitude(moveInput, 1f); } } public void OnPointerUp(PointerEventData eventData) { if (eventData.position.x <= Screen.width * 0.5f) { moveInput = Vector2.zero; } lookInput = Vector2.zero; } private void HandleMovement() { #if UNITY_EDITOR float h = Input.GetAxis("Horizontal"); float v = Input.GetAxis("Vertical"); if (h != 0 || v != 0) moveInput = new Vector2(h, v); #endif Vector3 move = transform.right * moveInput.x + transform.forward * moveInput.y; controller.Move(move * moveSpeed * Time.deltaTime); } private void HandleLook() { #if UNITY_EDITOR float mouseX = Input.GetAxis("Mouse X") * 2f; float mouseY = Input.GetAxis("Mouse Y") * 2f; if (mouseX != 0 || mouseY != 0) lookInput = new Vector2(mouseX, mouseY); #endif verticalRotation -= lookInput.y; verticalRotation = Mathf.Clamp(verticalRotation, -70f, 70f); if (cameraTransform) { cameraTransform.localRotation = Quaternion.Euler(verticalRotation, 0f, 0f); } transform.Rotate(Vector3.up * lookInput.x); lookInput = Vector2.zero; } private void CheckInteractables() { if (!cameraTransform) return; Ray ray = new Ray(cameraTransform.position, cameraTransform.forward); if (Physics.Raycast(ray, out RaycastHit hit, 3f)) { IInteractable interactable = hit.collider.GetComponent(); if (interactable != null) { currentInteractable = hit.collider.gameObject; if (interactionButton) interactionButton.SetActive(true); if (isInteracting) { interactable.Interact(); isInteracting = false; } return; } } currentInteractable = null; if (interactionButton) interactionButton.SetActive(false); } // Привязывается к большой кнопке действия на UI экрана iPhone public void TriggerInteractionButton() { isInteracting = true; Handheld.Vibrate(); // Тактильный отклик } # ========================================== # 2. СЮЖЕТНЫЙ СКРИПТ И АТМОСФЕРА ХОРРОРА # ========================================== private void InitializeGame() { if (tvStaticScreen) tvStaticScreen.SetActive(false); if (tvContentScreen) tvContentScreen.SetActive(false); if (roomLight) roomLight.intensity = 1.2f; if (interactionButton) interactionButton.SetActive(false); } private IEnumerator MasterHorrorSequence() { // 1-2. Игрок в темной комнате перед выключенным ТВ yield return new WaitForSeconds(8f); // 3. Свет начинает мигать yield return StartCoroutine(FlickerLightRoutine(5, 0.1f, 0.4f)); // 4. Ощущение присутствия за спиной (тихий шепот) PlaySFX(whisperClip); yield return new WaitForSeconds(4f); // 5. Телевизор сам включается ActivateTV(); yield return new WaitForSeconds(6f); // 6-7. Тревожный звук и полное отключение света TriggerPowerOutageSequence(); // 8. Ожидание, пока игрок спрячется в шкаф yield return new WaitUntil(() => isHiding); // 9-10. Прошло 5 секунд в укрытии, тревога стихает yield return new WaitForSeconds(5f); StopScaryAudio(); // 11. Игрок выходит — активируется квест с электрическим счетчиком yield return new WaitUntil(() => !isHiding); Debug.Log("Найдите электрический счетчик на стене и включите его!"); } private IEnumerator FlickerLightRoutine(int count, float minDelay, float maxDelay) { for (int i = 0; i < count; i++) { if (roomLight) roomLight.enabled = !roomLight.enabled; PlaySFX(switchClickClip); yield return new WaitForSeconds(Random.Range(minDelay, maxDelay)); } if (roomLight) roomLight.enabled = true; } private void ActivateTV() { if (tvStaticScreen) tvStaticScreen.SetActive(true); PlaySFX(tvStaticClip); StartCoroutine(ShowRoomOnTVDelayed()); } private IEnumerator ShowRoomOnTVDelayed() { yield return new WaitForSeconds(2.5f); if (tvStaticScreen) tvStaticScreen.SetActive(false); if (tvContentScreen) tvContentScreen.SetActive(true); } private void TriggerPowerOutageSequence() { isPowerOut = true; if (roomLight) roomLight.intensity = 0.05f; if (sfxSource) { sfxSource.clip = scaryBuildupClip; sfxSource.loop = true; sfxSource.Play(); } StartCoroutine(KillLightsCompletely()); } private IEnumerator KillLightsCompletely() { yield return new WaitForSeconds(1.5f); if (roomLight) roomLight.enabled = false; } private void StopScaryAudio() { if (sfxSource) sfxSource.Stop(); if (ambienceSource) ambienceSource.Play(); } private void PlaySFX(AudioClip clip) { if (sfxSource && clip) sfxSource.PlayOneShot(clip); } # ========================================== # 3. ВЗАИМОДЕЙСТВИЕ С ОБЪЕКТАМИ КОМНАТЫ # ========================================== // Реализация интерфейса IInteractable для интерактивных объектов комнаты, // если этот скрипт вешается непосредственно на них (выключатель, шкаф, счетчик). public void Interact() { switch (objectType) { case ObjectType.LightSwitch: if (!isPowerOut && roomLight) { roomLight.enabled = !roomLight.enabled; PlaySFX(switchClickClip); } break; case ObjectType.Closet: isHiding = !isHiding; if (isHiding && closetHidePoint) { transform.position = closetHidePoint.position; } break; case ObjectType.FuseBox: if (isPowerOut) { isPowerOut = false; if (roomLight) { roomLight.enabled = true; roomLight.intensity = 1.2f; } PlaySFX(switchClickClip); Debug.Log("Электричество восстановлено! Победа."); } break; case ObjectType.TV: // Интерактив с телевизором при необходимости break; } } } }
